    import capture_system_or_process_audio as cspa

    # List & filter active audio apps deterministically (or use cspa.find_app("Zoom")):
    apps = [a for a in cspa.discover().applications if a.active_audio is True]
    app_target = apps[0].with_gain(0.9) if apps else cspa.default_output()

    # Mix default microphone + app into one 16 kHz mono float32 stream for Whisper / STT:
    converter = cspa.OutputConverter(rate=16000, channels=1, sample_format="f32")

        [cspa.microphone(), app_target],

        silence_on_idle=True,

    use capture_system_or_process_audio::{
        CapabilityError, CaptureTarget, DefaultDeviceMode, Stop, capture_single,
    };
    use std::time::Duration;

    fn main() -> Result<(), CapabilityError> {
        let target = CaptureTarget::default_output(DefaultDeviceMode::Follow);
        let mut manager = capture_single(target)?;

        let fmt = manager.output_format().expect("active output format").clone();
        let mut pcm_buf = vec![0u8; 480 * fmt.bytes_per_frame()];

        // Zero-allocation steady-state pull into caller-owned buffer:
        let res = manager
            .read_into(&mut pcm_buf, 480, Duration::from_millis(200))
            .expect("read PCM frames");
        println!("Captured {} frames @ {} Hz", res.frames, fmt.rate());

        manager.stop(Stop::Drain);
        manager.wait_cleanup(Duration::from_secs(2));
        Ok(())
    }

    #include <cspa.hpp>
    #include <cstdio>
    #include <vector>

    int main() {
        auto speaker = cspa::Target::default_output(cspa::DefaultDeviceMode::Follow).take_value();
        auto mic     = cspa::Target::microphone(cspa::DefaultDeviceMode::Follow).take_value();

        cspa::MixerConfig mixer;
        mixer.set_silence_on_idle(true);

        cspa::CaptureRequest req({speaker, mic});
        req.set_mixer_config(mixer)
           .set_output_converter(cspa::OutputConverter(48000, 2, cspa::SampleFormat::F32));

        auto stream_res = cspa::ManagedStream::open(req);
        if (!stream_res.is_ok()) return 1;
        cspa::ManagedStream stream = stream_res.take_value();

        std::vector<uint8_t> pcm(480 * 2 * sizeof(float));
        auto rd = stream.read_into(pcm.data(), pcm.size(), 480, 200);
        if (rd.is_ok()) {
            std::printf("Captured %llu mixed frames\n", static_cast<unsigned long long>(rd.value().frames));
        }

        stream.stop(false);
        stream.wait_cleanup(2000);
        return 0;
    }

### Why `PyAudioWPatch` Hit an Architectural Wall — and Why This Took Nearly a Year to Build

As `PyAudioWPatch` grew, users repeatedly requested capabilities that were **architecturally impossible to implement** inside a legacy PortAudio C wrapper:

- **Per-application isolation & macOS / Linux support:** PortAudio models hardware endpoints only — it cannot attach to Windows per-process loopback trees, Linux PipeWire/PulseAudio application graphs, or macOS Core Audio Process Taps / ScreenCaptureKit without throwing away the underlying C engine (leaving macOS and Linux users stuck installing `BlackHole` or `VB-Cable`).
- **Real-world loopback traps that break naive recorders:** Silent WASAPI speakers emit zero packets (causing blocking `stream.read()` loops in `PyAudioWPatch` to hang forever); unplugging headphones breaks pinned streams; combining `[Microphone + App/System Audio]` drifts out of sync across independent hardware clocks; and Free-Threaded Python `3.14t`/`3.15t` (`-Xgil=0`) collides with PortAudio's global C state.

Rather than bolting fragile workarounds onto a 20-year-old C wrapper, **`capture-system-or-process-audio` is the result of nearly a year of dedicated ground-up development and polish in safe Rust** — turning years of `PyAudioWPatch` production feedback into one zero-steady-state-allocation engine across **Python, Rust, C++17, and C11**:

| Capability | `capture-system-or-process-audio` | `PyAudioWPatch` | `sounddevice` / `PyAudio` | `cpal` (Rust) | Virtual Cables (`VB-Cable` / `BlackHole`) |
| --- | :---: | :---: | :---: | :---: | :---: |
| **System Speaker Loopback** | ✅ Win · Linux · macOS | ✅ Windows only | ❌ | ⚠️ Partial | ⚠️ Manual driver install |
| **Single App / Process Tree Isolation** | ✅ Win · Linux · macOS | ❌ | ❌ | ❌ | ❌ |
| **Microphone + App/System Live Mix** | ✅ Built-in (`CspaMixer` + drift sync) | ❌ Manual | ❌ Manual | ❌ Manual | ⚠️ Manual multi-output |
| **Silent-Speaker Hang Protection** | ✅ (`silence_on_idle` + timeouts) | ❌ Hangs `read()` | ❌ | ❌ | N/A |
| **Auto-Follow Default Device Switches** | ✅ (`Follow` mode) | ❌ | ❌ | ❌ | ❌ Breaks volume keys |
| **Built-In Resampler & Format Converter** | ✅ (`16k`/`44.1k`/`48k`, `f32`/`i16`/`i24`/`i32`) | ❌ | ❌ | ❌ | ❌ |
| **OS Permission Preflight & Prompt API** | ✅ Honest 5-state report | ❌ | ❌ | ❌ | N/A |

## Platform & Backend Matrix

When a requested scope or capability is unsupported on the active OS or OS build, `capture-system-or-process-audio` **fails explicitly with a typed error (`CapabilityError` / `PlanError`)** — it **never** silently substitutes full-system audio for single-process capture.

| Platform | Native Backends | System & Device Output | Application / Process Capture | Microphones & Inputs |
| --- | --- | --- | --- | --- |
| **Windows 10 / 11** (`x64`, `ARM64`) | `wasapi` | ✅ Build `19041+` (`Follow`/`Pin`, `device`) | ✅ Build `20348+` (`scope="tree"`; `exact` rejected explicitly) | ✅ Build `19041+` (`Follow`/`Pin`, `input_device`) |
| **Linux** (`x86_64`, `aarch64`) | `pipewire` (`0.3.65+`), `pulseaudio` (`15.0+`) | ✅ Sink monitor & default sink tracking | ✅ PipeWire virtual-sink fan-in (`tree`/`exact`) & PulseAudio stream mix | ✅ Physical/virtual sources (`0` compile-time `NEEDED` links) |
| **macOS** (`Apple Silicon arm64`) | `coreaudio`, `screencapturekit` | ✅ `macOS 14.2+` Core Audio Process Taps | ✅ `macOS 14.2+` Core Audio Taps (`tree`/`exact`) & `macOS 13.0+` SCK (`exact`) | ✅ `macOS 13.0+` Core Audio HAL input (`AVMediaTypeAudio`) |
