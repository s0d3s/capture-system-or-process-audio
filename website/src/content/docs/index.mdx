    ```rust
    use capture_system_or_process_audio::{
        CapabilityError, CaptureTarget, DefaultDeviceMode, Stop, capture_single,
    };
    use std::time::Duration;

    fn main() -> Result<(), CapabilityError> {
        let target = CaptureTarget::default_output(DefaultDeviceMode::Follow);
        let mut manager = capture_single(target)?;

        let fmt = manager.output_format().expect("active output format").clone();
        let mut pcm_buf = vec![0u8; 480 * fmt.bytes_per_frame()];

        // Zero-allocation steady-state pull into caller-owned buffer:
        let res = manager
            .read_into(&mut pcm_buf, 480, Duration::from_millis(200))
            .expect("read PCM frames");
        println!("Captured {} frames @ {} Hz", res.frames, fmt.rate());

        manager.stop(Stop::Drain);
        manager.wait_cleanup(Duration::from_secs(2));
        Ok(())
    }
    ```

    ```cpp
    #include <cspa.hpp>
    #include <cstdio>
    #include <vector>

    int main() {
        auto speaker = cspa::Target::default_output(cspa::DefaultDeviceMode::Follow).take_value();
        auto mic     = cspa::Target::microphone(cspa::DefaultDeviceMode::Follow).take_value();

        cspa::MixerConfig mixer;
        mixer.set_silence_on_idle(true);

        cspa::CaptureRequest req({speaker, mic});
        req.set_mixer_config(mixer)
           .set_output_converter(cspa::OutputConverter(48000, 2, cspa::SampleFormat::F32));

        auto stream_res = cspa::ManagedStream::open(req);
        if (!stream_res.is_ok()) return 1;
        cspa::ManagedStream stream = stream_res.take_value();

        std::vector<uint8_t> pcm(480 * 2 * sizeof(float));
        auto rd = stream.read_into(pcm.data(), pcm.size(), 480, 200);
        if (rd.is_ok()) {
            std::printf("Captured %llu mixed frames\n", static_cast<unsigned long long>(rd.value().frames));
        }

        stream.stop(false);
        stream.wait_cleanup(2000);
        return 0;
    }
    ```

    ```c
            return 1;
        }

        cspa_stream *stream = NULL;
        if (cspa_stream_open(&target, CSPA_OVERFLOW_DROP_OLDEST, &stream) != CSPA_STATUS_OK) {

                   (unsigned long long)out.actual_frames,
                   (unsigned long long)out.copy_bytes);
        }

        uint32_t cleaned = 0;
        cspa_stream_wait_cleanup(stream, 2000u, &cleaned);
        cspa_stream_release(stream);
        return 0;
    }
    ```

## Why Beyond `PyAudioWPatch`?

Four years ago, [`@s0d3s`](https://github.com/s0d3s) created **[`PyAudioWPatch`](https://github.com/s0d3s/PyAudioWPatch)** to bring Windows WASAPI speaker loopback to Python's `PyAudio`. Today it is the standard Windows loopback package in the Python ecosystem — serving **140,000+ monthly PyPI downloads** (`~1.6M+` annually) and powering **340+ open-source repositories** across AI meeting copilots, live Whisper/STT transcription bots, voice agents, and desktop recorders.
